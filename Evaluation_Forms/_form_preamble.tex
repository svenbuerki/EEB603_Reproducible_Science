% Shared LaTeX preamble for all EEB 603 evaluation forms.

% A grey rule used to separate rubric sections
\newcommand{\sectionrule}{\vspace{-1mm}\textcolor{gray!55}{\rule{\textwidth}{0.4pt}}\vspace{1mm}}

\newcommand{\scoreline}[1]{%
